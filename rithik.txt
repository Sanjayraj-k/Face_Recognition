# Custom JSON encoder
class NumpyEncoder(json.JSONEncoder):
    def default(self, obj):
        if isinstance(obj, np.integer):
            return int(obj)
        if isinstance(obj, np.floating):
            return float(obj)
        if isinstance(obj, np.ndarray):
            return obj.tolist()
        if isinstance(obj, np.bool_):
            return bool(obj)
        if isinstance(obj, datetime):
            return obj.isoformat()
        return super(NumpyEncoder, self).default(obj)

# Helper function to save image to filesystem and return path
def save_image_to_filesystem(image_array, folder, filename_prefix, username):
    try:
        user_folder = os.path.join(folder, username)
        os.makedirs(user_folder, exist_ok=True)
        timestamp = datetime.utcnow().strftime('%Y%m%d_%H%M%S_%f')
        filename = secure_filename(f"{filename_prefix}_{timestamp}.jpg")
        filepath = os.path.join(user_folder, filename)
        cv2.imwrite(filepath, image_array)
        return filepath
    except Exception as e:
        print(f"Error saving image to filesystem: {str(e)}")
        return None

# Helper function to read image from filesystem and convert to base64
def image_to_base64(filepath):
    try:
        if os.path.exists(filepath):
            with open(filepath, "rb") as f:
                encoded_image = base64.b64encode(f.read()).decode('utf-8')
                return f"data:image/jpeg;base64,{encoded_image}"
        return None
    except Exception as e:
        print(f"Error reading image from {filepath}: {str(e)}")
        return None

# Face processing functions
def extract_faces(img_array, confidence_threshold=0.8):
    if img_array is None:
        return [], []
    rgb_img = cv2.cvtColor(img_array, cv2.COLOR_BGR2RGB)
    faces = detector.detect_faces(rgb_img)
    face_images, face_positions = [], []
    for face in faces:
        if face['confidence'] >= confidence_threshold:
            x, y, w, h = face['box']
            x1, y1, x2, y2 = max(0, x), max(0, y), min(rgb_img.shape[1], x + w), min(rgb_img.shape[0], y + h)
            face_img = rgb_img[y1:y2, x1:x2]
            if face_img.size > 0:
                face_images.append(face_img)
                face_positions.append((x1, y1, x2, y2))
    return face_images, face_positions

def extract_features(face_img):
    face_img = Image.fromarray(face_img)
    face_tensor = transform(face_img).unsqueeze(0).to(device)
    with torch.no_grad():
        embedding = facenet(face_tensor).cpu().numpy()
    return embedding / np.linalg.norm(embedding)

def get_file_hash(file_path):
    hasher = hashlib.md5()
    with open(file_path, 'rb') as f:
        buf = f.read(65536)
        while len(buf) > 0:
            hasher.update(buf)
            buf = f.read(65536)
    return hasher.hexdigest()

def extract_face_bounding_box(image_data, face_position):
    try:
        if not face_position or len(face_position) != 4:
            return image_data
        
        if image_data.startswith('data:image'):
            image_data = image_data.split(',')[1]
        
        image_bytes = base64.b64decode(image_data)
        nparr = np.frombuffer(image_bytes, np.uint8)
        img = cv2.imdecode(nparr, cv2.IMREAD_COLOR)
        
        if img is None:
            return image_data
        
        x1, y1, x2, y2 = face_position
        x1, y1, x2, y2 = int(x1), int(y1), int(x2), int(y2)
        
        h, w = img.shape[:2]
        x1, y1 = max(0, x1), max(0, y1)
        x2, y2 = min(w, x2), min(h, y2)
        
        if x2 > x1 and y2 > y1:
            face_img = img[y1:y2, x1:x2]
            _, buffer = cv2.imencode('.jpg', face_img)
            face_b64 = base64.b64encode(buffer).decode('utf-8')
            return f"data:image/jpeg;base64,{face_b64}"
        
        return image_data
    except Exception as e:
        print(f"Error extracting face bounding box: {str(e)}")
        return image_data

# Cache functions
def load_cache(username):
    embeddings = embeddings_collection.find({"username": username})
    cache = {}
    for emb in embeddings:
        cache[emb['filepath']] = {
            'hash': emb['hash'],
            'faces': emb['faces']
        }
    print(f"Cache loaded with {len(cache)} entries for user {username}.")
    return cache

def save_cache(username, cache):
    global cache_last_updated
    try:
        embeddings_collection.delete_many({"username": username})
        for filepath, data in cache.items():
            embeddings_collection.insert_one({
                "username": username,
                "filepath": filepath,
                "hash": data['hash'],
                "faces": data['faces'],
                "last_updated": datetime.utcnow()
            })
        cache_last_updated = time.time()
        print(f"Cache saved with {len(cache)} entries for user {username}.")
        return True
    except Exception as e:
        print(f"Error saving cache: {str(e)}")
        return False

def check_album_changes(username, cache):
    user_photos = photos_collection.find({"username": username})
    photo_paths = {photo['filepath'] for photo in user_photos}
    if len(photo_paths) != len(cache):
        return True
    for img_path in photo_paths:
        file_hash = get_file_hash(img_path)
        if img_path not in cache or cache[img_path]['hash'] != file_hash:
            return True
    return False

def update_cache_async(username):
    global cache_updating
    if cache_updating:
        return
    cache_updating = True
    thread = Thread(target=update_cache, args=(username,))
    thread.daemon = True
    thread.start()

def update_cache(username):
    global cache_updating
    try:
        print(f"Starting cache update for user {username}...")
        user_photos = photos_collection.find({"username": username})
        supported_extensions = ['jpg', 'jpeg', 'png']
        new_cache = {}
        for photo in user_photos:
            img_path = photo['filepath']
            if not any(img_path.lower().endswith(ext) for ext in supported_extensions):
                continue
            file_hash = get_file_hash(img_path)
            if img_path in new_cache and new_cache[img_path]['hash'] == file_hash:
                continue
            try:
                img_array = cv2.imread(img_path)
                if img_array is None:
                    continue
                faces, positions = extract_faces(img_array)
                if not faces:
                    continue
                face_data = []
                for face, position in zip(faces, positions):
                    features = extract_features(face)
                    face_data.append({
                        'embedding': features.tolist(),
                        'position': position
                    })
                new_cache[img_path] = {
                    'hash': file_hash,
                    'faces': face_data
                }
            except Exception as e:
                print(f"Error processing {img_path}: {str(e)}")
                continue
        save_cache(username, new_cache)
        print(f"Cache update completed for user {username}.")
    except Exception as e:
        print(f"Error updating cache: {str(e)}")
    finally:
        cache_updating = False

def find_matches_in_album(username, solo_embedding, similarity_threshold=0.3):
    matches = []
    cache = load_cache(username)
    for img_path, cache_entry in cache.items():
        try:
            if 'faces' not in cache_entry or not cache_entry['faces']:
                continue
            matched = False
            best_similarity = 1.0
            best_face_position = None
            for face_data in cache_entry['faces']:
                features = np.array(face_data['embedding'])
                position = face_data['position']
                similarity = float(cosine(solo_embedding.flatten(), features.flatten()))
                if similarity < best_similarity:
                    best_similarity = similarity
                    best_face_position = position
                if similarity < similarity_threshold:
                    matched = True
            if matched:
                similarity_percentage = float((1 - best_similarity) * 100)
                if similarity_percentage <= 70:
                    continue
                img_array = cv2.imread(img_path)
                if img_array is None:
                    continue
                result_img = img_array.copy()
                if best_face_position:
                    x1, y1, x2, y2 = best_face_position
                    cv2.rectangle(result_img, (x1, y1), (x2, y2), (0, 255, 0), 2)
                
                # Save images to filesystem
                highlighted_filepath = save_image_to_filesystem(result_img, app.config['CACHE_FOLDER'], f"highlighted_{os.path.basename(img_path)}", username)
                original_filepath = save_image_to_filesystem(img_array, app.config['CACHE_FOLDER'], f"original_{os.path.basename(img_path)}", username)
                
                if highlighted_filepath and original_filepath:
                    filename = os.path.basename(img_path)
                    matches.append({
                        "filename": filename,
                        "filepath": img_path,
                        "similarity": similarity_percentage,
                        "highlighted_filepath": highlighted_filepath,
                        "original_filepath": original_filepath
                    })
        except Exception as e:
            print(f"Error processing cached entry {img_path}: {str(e)}")
            continue
    matches.sort(key=lambda x: x["similarity"], reverse=True)
    return matches

@app.route('/api/login', methods=['POST'])
def login():
    data = request.get_json()
    if not data:
        return jsonify({"success": False, "message": "Invalid request data"}), 400
    username = data.get('username')
    password = data.get('password')
    if not username or not password:
        return jsonify({"success": False, "message": "Username and password required"}), 400
    user = users_collection.find_one({"username": username})
    if user and bcrypt.checkpw(password.encode('utf-8'), user['password']):
        session['username'] = username
        return jsonify({"success": True, "message": "Logged in successfully", "username": username})
    return jsonify({"success": False, "message": "Invalid credentials"}), 401

@app.route('/api/upload_album', methods=['POST'])
def upload_album():
    if 'username' not in session:
        return jsonify({"success": False, "message": "Please log in first"}), 401
    
    username = session['username']
    
    if 'album_photos' not in request.files:
        return jsonify({"success": False, "message": "No files uploaded"}), 400
    
    files = request.files.getlist('album_photos')
    if not files or all(file.filename == '' for file in files):
        return jsonify({"success": False, "message": "No files selected"}), 400
    
    uploaded_files = []
    supported_extensions = ['jpg', 'jpeg', 'png']
    
    for file in files:
        if file and file.filename:
            filename = secure_filename(file.filename)
            if not any(filename.lower().endswith(ext) for ext in supported_extensions):
                continue
            
            temp_path = os.path.join(app.config['UPLOAD_FOLDER'], f"temp_{filename}")
            file.save(temp_path)
            
            file_hash = get_file_hash(temp_path)
            
            existing_photo = photos_collection.find_one({"username": username, "file_hash": file_hash})
            if existing_photo:
                os.remove(temp_path)
                continue
            
            user_folder = os.path.join(app.config['ALBUM_FOLDER'], username)
            os.makedirs(user_folder, exist_ok=True)
            
            filepath = os.path.join(user_folder, filename)
            os.rename(temp_path, filepath)
            
            photos_collection.insert_one({
                "username": username,
                "filename": filename,
                "filepath": filepath,
                "file_hash": file_hash,
                "uploaded_at": datetime.utcnow()
            })
            
            uploaded_files.append(filename)
    
    if uploaded_files:
        update_cache_async(username)
        return jsonify({
            "success": True, 
            "message": f"Successfully uploaded {len(uploaded_files)} photos",
            "uploaded_files": uploaded_files
        })
    else:
        return jsonify({"success": False, "message": "No valid image files uploaded"}), 400

@app.route('/api/search', methods=['POST'])
def search():
    if 'username' not in session:
        return jsonify({"success": False, "message": "Please log in first"}), 401
    
    username = session['username']
    
    if 'solo_photo' not in request.files:
        return jsonify({"success": False, "message": "No file uploaded"}), 400
    
    file = request.files['solo_photo']
    if file.filename == '':
        return jsonify({"success": False, "message": "No file selected"}), 400
    
    try:
        filename = secure_filename(file.filename)
        temp_path = os.path.join(app.config['UPLOAD_FOLDER'], f"temp_{filename}")
        file.save(temp_path)
        
        img_array = cv2.imread(temp_path)
        if img_array is None:
            os.remove(temp_path)
            return jsonify({"success": False, "message": "Invalid image file"}), 400
        
        faces, positions = extract_faces(img_array)
        if not faces:
            os.remove(temp_path)
            return jsonify({"success": False, "message": "No faces detected in the image"}), 400
        
        solo_embedding = extract_features(faces[0])
        
        matches = find_matches_in_album(username, solo_embedding)
        
        os.remove(temp_path)
        
        if matches:
            return jsonify({
                "success": True,
                "match_found": True,
                "message": f"Found {len(matches)} matching photos",
                "matches": matches
            })
        else:
            return jsonify({
                "success": True,
                "match_found": False,
                "message": "No matching photos found",
                "matches": []
            })
    
    except Exception as e:
        return jsonify({"success": False, "message": f"Search failed: {str(e)}"}), 500

@app.route('/api/get_person_clusters', methods=['POST'])
def get_person_clusters():
    if 'username' not in session:
        return jsonify({"success": False, "message": "Please log in first"}), 401
    
    username = session['username']
    try:
        data = request.get_json()
        photos = data.get('photos', [])
        
        if not photos:
            return jsonify({"success": False, "message": "No photos provided"}), 400
        
        cache = load_cache(username)
        clusters = {}
        cluster_id = 0
        
        for photo in photos:
            filepath = photo.get('filepath', '')
            if filepath not in cache or 'faces' not in cache[filepath]:
                continue
                
            face_embeddings = cache[filepath]['faces']
            if not face_embeddings:
                continue
            
            person_embedding = np.array(face_embeddings[0]['embedding'])
            
            assigned_cluster = None
            for cluster in clusters.values():
                if cluster['representative_embedding'] is not None:
                    similarity = float(cosine(person_embedding.flatten(), cluster['representative_embedding'].flatten()))
                    if similarity < 0.3:
                        assigned_cluster = cluster
                        break
            
            if assigned_cluster is None:
                cluster_id += 1
                assigned_cluster = {
                    'id': f'person_{cluster_id}',
                    'name': f'Person {cluster_id}',
                    'personId': f'person_{cluster_id}',
                    'photos': [],
                    'faceImage': None,
                    'representative_embedding': person_embedding,
                    'avgSimilarity': 0
                }
                clusters[cluster_id] = assigned_cluster
            
            assigned_cluster['photos'].append(photo)
            
            if assigned_cluster['faceImage'] is None:
                assigned_cluster['faceImage'] = photo.get('image_data', '')
        
        for cluster in clusters.values():
            if cluster['photos']:
                similarities = []
                for photo in cluster['photos']:
                    if 'similarity' in photo:
                        similarities.append(photo['similarity'])
                
                if similarities:
                    cluster['avgSimilarity'] = round(sum(similarities) / len(similarities))
                else:
                    cluster['avgSimilarity'] = 95
        
        cluster_list = list(clusters.values())
        cluster_list.sort(key=lambda x: len(x['photos']), reverse=True)
        
        return jsonify({
            "success": True,
            "clusters": cluster_list,
            "message": f"Found {len(cluster_list)} person clusters"
        })
    
    except Exception as e:
        return jsonify({"success": False, "message": f"Error creating person clusters: {str(e)}"}), 500

@app.route('/api/get_persistent_clusters', methods=['GET'])
def get_persistent_clusters():
    if 'username' not in session:
        return jsonify({"success": False, "message": "Please log in first"}), 401

@app.route('/api/add_matches_to_clusters', methods=['POST'])
def add_matches_to_clusters():
    if 'username' not in session:
        return jsonify({"success": False, "message": "Please log in first"}), 401
    
    username = session['username']
    try:
        data = request.get_json()
        matches = data.get('matches', [])
        existing_clusters = data.get('existingClusters', [])
        
        print(f"Processing {len(matches)} matches for clustering")
        
        cache = load_cache(username)
        print(f"Cache loaded with {len(cache)} entries")
        
        updated_clusters = existing_clusters.copy() if existing_clusters else []
        
        for i, match in enumerate(matches):
            print(f"Processing match {i+1}: {match.get('filename', 'unknown')}")
            
            filepath = match.get('filepath', '')
            if filepath not in cache:
                print(f"Filepath {filepath} not found in cache")
                continue
                
            if 'faces' not in cache[filepath]:
                print(f"No faces found for {filepath}")
                continue
                
            face_embeddings = cache[filepath]['faces']
            if not face_embeddings:
                print(f"No face embeddings for {filepath}")
                continue
            
            person_embedding = np.array(face_embeddings[0]['embedding'])
            face_position = face_embeddings[0].get('position', [])
            
            print(f"Face position: {face_position}")
            
            assigned_cluster = None
            for cluster in updated_clusters:
                if cluster.get('representative_embedding') is not None:
                    similarity = float(cosine(person_embedding.flatten(), np.array(cluster['representative_embedding']).flatten()))
                    print(f"Similarity with {cluster['name']}: {similarity}")
                    if similarity < 0.3:
                        assigned_cluster = cluster
                        print(f"Assigned to existing cluster: {cluster['name']}")
                        break
            
            if assigned_cluster is None:
                cluster_id = f"person_{len(updated_clusters) + 1}"
                
                # Use highlighted_filepath if available, else original_filepath
                face_profile_image = match.get('highlighted_filepath', match.get('original_filepath', ''))
                if face_profile_image:
                    face_profile_image = image_to_base64(face_profile_image) or ''
                
                assigned_cluster = {
                    'id': cluster_id,
                    'name': f'Person {len(updated_clusters) + 1}',
                    'personId': cluster_id,
                    'photos': [],
                    'faceImage': face_profile_image,
                    'representative_embedding': person_embedding.tolist(),
                    'avgSimilarity': 95
                }
                updated_clusters.append(assigned_cluster)
                print(f"Created new cluster: {cluster_id}")
            
            photo_exists = any(photo.get('filename') == match.get('filename') for photo in assigned_cluster['photos'])
            if not photo_exists:
                assigned_cluster['photos'].append(match)
                print(f"Added photo to cluster {assigned_cluster['name']}")
            else:
                print(f"Photo already exists in cluster {assigned_cluster['name']}")
        
        print(f"Final clusters: {len(updated_clusters)}")
        for cluster in updated_clusters:
            print(f"Cluster {cluster['name']}: {len(cluster['photos'])} photos")
        
        person_clusters_collection.delete_many({"username": username})
        for cluster in updated_clusters:
            face_image_filepath = ''
            if cluster.get('faceImage', '').startswith('data:image'):
                image_data = cluster['faceImage'].split(',')[1]
                image_bytes = base64.b64decode(image_data)
                nparr = np.frombuffer(image_bytes, np.uint8)
                img = cv2.imdecode(nparr, cv2.IMREAD_COLOR)
                if img is not None:
                    face_image_filepath = save_image_to_filesystem(img, app.config['TAGGED_IMAGES_FOLDER'], f"cluster_{cluster['id']}", username)
            
            person_clusters_collection.insert_one({
                "username": username,
                "cluster_id": cluster['id'],
                "person_name": cluster['name'],
                "person_id": cluster['personId'],
                "photos": cluster.get('photos', []),
                "face_image_filepath": face_image_filepath,
                "avg_similarity": cluster.get('avgSimilarity', 95),
                "representative_embedding": cluster.get('representative_embedding', []),
                "created_at": datetime.utcnow(),
                "updated_at": datetime.utcnow()
            })
        
        return jsonify({
            "success": True,
            "clusters": updated_clusters,
            "message": f"Updated clusters with {len(matches)} new photos"
        })
    
    except Exception as e:
        print(f"Error in add_matches_to_clusters: {str(e)}")
        return jsonify({"success": False, "message": f"Error adding matches to clusters: {str(e)}"}), 500

@app.route('/api/save_tagged_image', methods=['POST'])
def save_tagged_image():
    if 'username' not in session:
        return jsonify({"success": False, "message": "Please log in first"}), 401
    
    username = session['username']
    try:
        data = request.get_json()
        tagged_image = data.get('taggedImage', {})
        
        print(f"Saving tagged image for user {username}: {tagged_image.get('tagName', 'Unknown')}")
        
        if not tagged_image:
            return jsonify({"success": False, "message": "No tagged image data provided"}), 400
        
        # Save tagged image to filesystem
        image_data_filepath = ''
        if tagged_image.get('imageData', '').startswith('data:image'):
            image_data = tagged_image['imageData'].split(',')[1]
            image_bytes = base64.b64decode(image_data)
            nparr = np.frombuffer(image_bytes, np.uint8)
            img = cv2.imdecode(nparr, cv2.IMREAD_COLOR)
            if img is not None:
                image_data_filepath = save_image_to_filesystem(img, app.config['TAGGED_IMAGES_FOLDER'], f"tagged_{tagged_image.get('id')}", username)
        
        search_results = tagged_image.get('searchResults', [])
        processed_search_results = []
        
        for result in search_results:
            processed_result = {
                "filename": result.get('filename', ''),
                "filepath": result.get('filepath', ''),
                "similarity": result.get('similarity', 0),
                "highlighted_filepath": result.get('highlighted_filepath', ''),
                "original_filepath": result.get('original_filepath', result.get('highlighted_filepath', ''))
            }
            processed_search_results.append(processed_result)
        
        document = {
            "username": username,
            "tag_id": tagged_image.get('id'),
            "tag_name": tagged_image.get('tagName'),
            "image_data_filepath": image_data_filepath,
            "search_results": processed_search_results,
            "created_at": datetime.utcnow(),
            "updated_at": datetime.utcnow()
        }
        
        print(f"Document to save: {document}")
        print(f"Search results count: {len(processed_search_results)}")
        
        result = tagged_images_collection.insert_one(document)
        print(f"Inserted document with ID: {result.inserted_id}")
        
        return jsonify({
            "success": True,
            "message": f"Tagged image '{tagged_image.get('tagName')}' saved successfully"
        })
    
    except Exception as e:
        print(f"Error saving tagged image for user {username}: {str(e)}")
        return jsonify({"success": False, "message": f"Error saving tagged image: {str(e)}"}), 500

@app.route('/api/delete_tagged_image', methods=['DELETE'])
def delete_tagged_image():
    if 'username' not in session:
        return jsonify({"success": False, "message": "Please log in first"}), 401
    
    username = session['username']
    try:
        data = request.get_json()
        tag_id = data.get('tagId')
        
        if not tag_id:
            return jsonify({"success": False, "message": "Tag ID required"}), 400
        
        tagged_image = tagged_images_collection.find_one({"username": username, "tag_id": tag_id})
        if not tagged_image:
            return jsonify({"success": False, "message": "Tagged image not found"}), 404
        
        # Delete associated image files
        if tagged_image.get('image_data_filepath') and os.path.exists(tagged_image['image_data_filepath']):
            os.remove(tagged_image['image_data_filepath'])
        for result in tagged_image.get('search_results', []):
            for filepath in [result.get('highlighted_filepath', ''), result.get('original_filepath', '')]:
                if filepath and os.path.exists(filepath):
                    os.remove(filepath)
        
        result = tagged_images_collection.delete_one({"username": username, "tag_id": tag_id})
        
        if result.deleted_count == 0:
            return jsonify({"success": False, "message": "Tagged image not found"}), 404
        
        return jsonify({
            "success": True,
            "message": "Tagged image deleted successfully"
        })
    
    except Exception as e:
        return jsonify({"success": False, "message": f"Error deleting tagged image: {str(e)}"}), 500
